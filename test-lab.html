Playwright Test Script

portfolio.spec.ts
import { test, expect, Page } from '@playwright/test';

test.describe('Portfolio Element Coverage', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/test-lab.html');
    await page.waitForLoadState('domcontentloaded');
  });

  test('text inputs', async ({ page }) => {
    await page.getByTestId('input-text').fill('Hello Playwright');
    await expect(page.getByTestId('input-text')).toHaveValue('Hello Playwright');
  });

  test('checkboxes', async ({ page }) => {
    await page.getByTestId('checkbox-agree').check();
    await expect(page.getByTestId('checkbox-agree')).toBeChecked();
  });

  test('select dropdown', async ({ page }) => {
    await page.getByTestId('select-country').selectOption('india');
    await expect(page.getByTestId('select-country')).toHaveValue('india');
  });

  test('buttons', async ({ page }) => {
    await expect(page.getByTestId('btn-disabled')).toBeDisabled();
    await expect(page.getByTestId('btn-primary')).toBeEnabled();
  });

  test('dialog', async ({ page }) => {
    await page.getByTestId('btn-open-dialog').click();
    await expect(page.getByTestId('dialog-modal')).toBeVisible();
    await page.getByTestId('btn-close-dialog').click();
  });

  test('table', async ({ page }) => {
    await expect(page.getByTestId('cell-name-1')).toHaveText('Harsha');
    await expect(page.getByTestId('cell-role-1')).toHaveText('QA Automation Engineer/SDET');
    const rows = page.locator('[data-testid="table-data"] tbody tr');
    await expect(rows).toHaveCount(3);
  });

  test('form submission', async ({ page }) => {
    await page.getByTestId('form-input-name').fill('Test User');
    await page.getByTestId('form-btn-submit').click();
    await expect(page.getByTestId('form-status')).toHaveText('Submitted');
  });
});
